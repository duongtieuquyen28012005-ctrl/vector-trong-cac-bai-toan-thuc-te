    \chapter{Toán thực tế xác định toạ độ điểm, toạ độ vectơ}
    \section{Một số điểm lý thuyết quan trọng}
    \subsection{Tọa độ điểm trong không gian $Oxyz$}
    \begin{tc}
    Một điểm $M(x;y;z)$ được xác định bởi ba khoảng cách có hướng:
        \begin{itemize}
            \item $x$: khoảng cách theo trục $Ox$ ($d(M,Ox)=|x|$)
            \item $y$: khoảng cách theo trục $Oy$ ($d(M,Oy)=|y|$)
            \item $z$: khoảng cách theo trục $Oz$ ($d(M,Oz)=|z|$)
        \end{itemize}
    \end{tc}
    \begin{nx} Khi chọn hệ trục phù hợp, các đỉnh của hình lập phương, hình hộp chữ nhật, hình lăng trụ,... thường có tọa độ nguyên (dễ tính).
    \end{nx}
    \subsection{Biểu diễn vectơ và các phép toán vectơ cơ bản}
    \begin{dn} Trong không gian với hệ tọa độ $Oxyz$, cho hai điểm
    $A(x_A;y_A;z_A)$ và $B(x_B;y_B;z_B)$. Khi đó, $\overrightarrow{AB}=(x_B-x_A;y_B-y_A;z_B-z_A).$
    
    Nếu $\vec{u}=(x_1;y_1;z_1)$ và $\vec{v}=(x_2;y_2;z_2)$, thì
    
    \hspace{4 cm} $\vec{u}+\vec{v}=(x_1+x_2;y_1+y_2;z_1+z_2);$
    
    \hspace{4 cm} $\vec{u}-\vec{v}=(x_1-x_2;y_1-y_2;z_1-z_2);$

    \hspace{4 cm} $m\vec{u}=(mx_1;my_1;mz_1) \quad \text{với } m\in\mathbb{R}.$
    \end{dn}
    \begin{nx} Hai vectơ $\vec{u}=(x_1;y_1;z_1)$,
    $\vec{v}=(x_2;y_2;z_2)$ ($\vec{v}\ne\vec{0}$) cùng phương khi và chỉ khi
    có một số thực $m$ sao cho
    $\small \begin{cases}
    x_1=mx_2,\\
    y_1=my_2,\\
    z_1=mz_2.
    \end{cases}$
    \end{nx}
    \subsection{Toạ độ các điểm đặc biệt}
    \begin{nx}
    Ta có một số điểm đặt biệt với toạ độ như sau:
        \begin{itemize}
            \item Nếu $M$ là trung điểm của đoạn thẳng $AB$ thì: 
            $$M\displaystyle \left(\frac{x_A+x_B}{2};\frac{y_A+y_B}{2};\frac{z_A+z_B}{2}\right)$$
            \item Nếu $G$ là trọng tâm của tam giác $ABC$ thì: 
            $$G\displaystyle \left(\frac{x_A+x_B+x_C}{3};\frac{y_A+y_B+y_C}{3};\frac{z_A+z_B+z_C}{3}\right)$$
            \item Nếu $M$ chia đoạn thẳng $AB$ theo tỉ số $k$, tức là 
            $\overrightarrow{MA}=k\overrightarrow{MB}$ thì: 
            $$M \displaystyle \left( \frac{x_A-kx_B}{1-k};\frac{y_A-ky_B}{1-k};\frac{z_A-kz_B}{1-k} \right)$$
        \end{itemize}
    \end{nx}
    \subsection{Bài toán có hướng Đông - Tây - Nam - Bắc}
    \begin{nx}
        Ta có một số điều cần lưu ý khi giải toán dạng này:
        \begin{itemize}
        \item Đặt hệ trục hợp lý, chẳng hạn như trong hình vẽ theo đúng thứ tự hướng.

        \begin{center}
            \includegraphics[width=0.9\textwidth]{Hinh1.jpg}
        \end{center}
    
        \item Mặt phẳng $(Oxy)$: trùng với mặt đất.
    
        \item Trục $Oz$: vuông góc mặt đất, hướng lên trên.
    
        \item Trục $Ox$: chọn theo hướng Đông -- Tây.
    
        \item Trục $Oy$: chọn theo hướng Bắc -- Nam.
    \end{itemize}
    
    Khi đó:
    \begin{itemize}
        \item Một điểm cách $a$ m về phía Đông thì hoành độ $x=+a$.
        \item Một điểm cách $a$ m về phía Tây thì hoành độ $x=-a$.
        \item Một điểm cách $a$ m về phía Bắc thì tung độ $y=+a$.
        \item Một điểm cách $a$ m về phía Nam thì tung độ $y=-a$.
    \end{itemize}
    \subsection{Một số kinh nghiệm khi làm bài}
    \begin{nx} Đối với các dạng toán thực tế xác định toạ độ điểm, toạ độ vectơ, ta cần:
    \begin{itemize} 
        \item Đặt hệ trục hợp lý: Thường đặt $O$ tại một góc, các trục song song với cạnh, đáy.
        \item Xác định tọa độ các đỉnh trước (từ dữ kiện chiều dài, rộng, cao).
        \item Viết vectơ cần tìm theo tọa độ.
    \end{itemize}
    \end{nx}
    \end{nx}
    \section{Một số ví dụ điển hình} 
    \begin{vd}
        Một ngôi nhà có dạng hình hộp chữ nhật, chiều cao $4$ m, mặt tiền $10$ m và chiều dài $20 m$. Trong nhà có một con nhện ở vị trí $A$, cách mặt đất $3m$, cách $2$ bức tường lần lượt là $2m$ và $1m$  (như hình minh họa).
        \begin{center}
            \includegraphics[width=0.9 \textwidth]{Hinh11.jpg}
        \end{center}
        Con nhện di chuyển theo đường thẳng đến góc nhà ở điểm $B$ với vận tốc không đổi $5$ cm/s.
        Thời gian con nhện di chuyển là bao nhiêu giây (làm tròn tới hàng đơn vị).
    \end{vd}
    \noindent \textbf{\underline{Nhận xét của tác giả.}}
        Ví dụ trên được lựa chọn vì đây là một tình huống khá trực quan, gần gũi với thực tế và có thể dễ dàng mô hình hóa bằng hệ trục tọa độ trong không gian. Từ vị trí của con nhện trong ngôi nhà, ta có thể chuyển các khoảng cách trong thực tế thành tọa độ của một điểm trong hệ trục $Oxyz$. Qua đó, bài toán giúp minh họa rõ cách sử dụng tọa độ điểm và công thức tính khoảng cách giữa hai điểm trong không gian để giải quyết một vấn đề thực tế.
    \noindent \textbf{\underline{Ý tưởng giải.}}
        Chọn hệ trục tọa độ $Oxyz$ phù hợp với kích thước của ngôi nhà, từ đó xác định tọa độ của vị trí con nhện và góc nhà. Tính độ dài đoạn thẳng nối hai vị trí bằng công thức khoảng cách trong không gian. Sau đó, sử dụng công thức $t=\dfrac{s}{v}$ để tính thời gian con nhện di chuyển.
    
    \noindent \underline{\textbf{Lời giải cụ thể:}}
    
    \noindent \textit{Bước 1. Thiết lập hệ trục và tọa độ}
        \begin{itemize}
            \item Đặt $A$ làm gốc tọa độ; các trục $Ox$, $Oy$, $Oz$ lần lượt cùng hướng với chiều dài, chiều rộng và chiều cao của ngôi nhà; mỗi đơn vị trên hệ trục tọa độ ứng với $1m$.
            
            \item Góc nhà ở điểm $B$ có tọa độ $B(0;10;4)$.
            
            \item Con nhện ở điểm $A$ cách mặt đất $3m$, cách $2$ bức tường lần lượt là $2m$ và $1m$ $\Rightarrow A(19;2;3)$.
        \end{itemize}

        \noindent \textit{Bước 2. Tính thời gian di chuyển}
        \begin{itemize}
            \item Quãng đường con nhện di chuyển là:
            \[
            AB=\sqrt{(0-19)^2+(10-2)^2+(4-3)^2}
            =\sqrt{426}\,(\mathrm{m})
            =100\sqrt{426}\,(\mathrm{cm}).
            \]
        
            \item Vậy thời gian con nhện di chuyển là:
            \[
            t=\frac{AB}{v}
            =\frac{100\sqrt{426}}{5}
            =20\sqrt{426}\approx 413\,(\mathrm{s}).
            \]
        \end{itemize}
        \noindent \underline{\textbf{Lời khuyên:}} Khi gặp các bài toán thực tế liên quan đến vị trí của các đối tượng trong không gian, nên lựa chọn hệ trục tọa độ sao cho các trục phù hợp với các hướng và kích thước đã cho. Sau khi xác định được tọa độ của các điểm, có thể sử dụng công thức tính khoảng cách trong không gian để đưa bài toán thực tế về một bài toán hình học tọa độ quen thuộc.
        \begin{vd}
            Một người đứng ở mặt đất điều khiển hai flycam để phục vụ trong một chương trình của đài truyền hình. Flycam I ở vị trí $A$ cách vị trí điều khiển $150\,\mathrm{m}$ về phía nam và $200\,\mathrm{m}$ về phía đông, đồng thời cách mặt đất $50\,\mathrm{m}$. Flycam II ở vị trí $B$ cách vị trí điều khiển $180\,\mathrm{m}$ về phía bắc và $240\,\mathrm{m}$ về phía tây, đồng thời cách mặt đất $60\,\mathrm{m}$. Chọn hệ trục tọa độ $Oxyz$ với gốc $O$ là vị trí người điều khiển, mặt phẳng $(Oxy)$ trùng với mặt đất, trục $Ox$ có hướng trùng với hướng nam, trục $Oy$ trùng với hướng đông, trục $Oz$ vuông góc với mặt đất hướng lên bầu trời, đơn vị trên mỗi trục tính theo mét. Khoảng cách giữa hai flycam đó bằng bao nhiêu mét (làm tròn đến hàng đơn vị)?
        \end{vd}
        \noindent \textbf{\underline{Nhận xét của tác giả}.}
        Ví dụ này được lựa chọn vì có sự liên hệ trực tiếp giữa các phương hướng quen thuộc trong thực tế như Đông, Tây, Nam, Bắc với các chiều của hệ trục tọa độ trong không gian. Tình huống điều khiển Flycam cũng cho thấy hệ tọa độ $Oxyz$ có thể được sử dụng để xác định vị trí của các đối tượng dựa trên khoảng cách theo từng phương và độ cao so với mặt đất. Qua đó, bài toán minh họa rõ cách chuyển một tình huống thực tế thành mô hình tọa độ trong không gian.
        
        \noindent \textbf{\underline{Ý tưởng giải.}}
        Dựa vào các phương hướng đã cho, lựa chọn hệ trục tọa độ sao cho các trục tọa độ lần lượt phù hợp với hướng Nam, Đông và phương thẳng đứng. Từ khoảng cách của mỗi Flycam đến các phương và độ cao so với mặt đất, xác định tọa độ của hai Flycam. Sau đó, sử dụng công thức tính khoảng cách giữa hai điểm trong không gian để tìm khoảng cách giữa chúng.
        
        \noindent \underline{\textbf{Lời giải cụ thể:}}
        \begin{center}
            \includegraphics[width=0.55\textwidth]{Hinh13.jpg}
        \end{center}
        
        \noindent \textit{Bước 1. Xác định tọa độ các điểm}
        \begin{itemize}
            \item Trục $Ox$ có hướng trùng với hướng Nam, trục $Oy$ trùng với hướng Đông, trục $Oz$ vuông góc với mặt đất hướng lên bầu trời. Khi đó:
            \begin{itemize}
                \item Flycam I ở vị trí $A$ cách vị trí điều khiển $150\,\mathrm{m}$ về phía Nam, $200\,\mathrm{m}$ về phía Đông và cách mặt đất $50\,\mathrm{m}\Rightarrow A(150;200;50)$.
                \item Flycam II ở vị trí $B$ cách vị trí điều khiển $180\,\mathrm{m}$ về phía Bắc, $240\,\mathrm{m}$ về phía Tây, đồng thời cách mặt đất $60\,\mathrm{m}\Rightarrow B(-180;-240;60)$.
            \end{itemize}
        \end{itemize}
        \noindent \textit{Bước 2. Xác định khoảng cách giữa hai Flycam}
        \begin{itemize}
            \item Khoảng cách giữa hai Flycam đó chính là khoảng cách giữa $A$ và $B$ và bằng: $d=AB=\sqrt{(150+180)^2+(200+240)^2+(50-60)^2}\approx550\,\mathrm{m}.$
        \end{itemize}
        \noindent \underline{\textbf{Lời khuyên:}} Khi mô hình hóa các bài toán thực tế có liên quan đến phương hướng, cần xác định rõ mối quan hệ giữa các hướng Đông, Tây, Nam, Bắc và chiều dương, chiều âm của các trục tọa độ. Nếu chọn chiều dương của một trục theo hướng Đông thì hướng Tây sẽ tương ứng với chiều âm; tương tự, nếu chiều dương được chọn theo hướng Nam thì hướng Bắc sẽ tương ứng với chiều âm. Ngoài ra, cần phân biệt các khoảng cách theo phương ngang với độ cao theo phương thẳng đứng để xác định chính xác tọa độ của các đối tượng.
        \begin{vd}
            Theo phong tục xa xưa của ông cha ta, người ta thường tổ chức lễ đốt đèn trời trong ngày Tết, ngày lễ còn có hẳn y tâm linh xua đuổi bóng đêm và ma quỷ. Người dân luôn nuôi dưỡng đức tin rằng ánh sáng và hơi ấm của đèn trời sẽ xua tan đi tà khí và những điều xấu xa. Đèn trời được làm từ củi bàng giấy và các nan tre đan lại với nhau. Quy trình thả đèn trời cũng khá đơn giản và nhanh chóng, chỉ cần đốt lửa rồi thả cho bay lên trời. 
            \begin{center}
            \includegraphics[width=0.4 \textwidth]{Hinh23.jpg}
            \end{center}
            Hai chiếc đèn trời được thả lên cùng một địa điểm. Sau một khoảng thời gian, chiếc thứ nhất nằm cách địa điểm xuất phát $200\,\mathrm{m}$ về hướng Đông và $200\,\mathrm{m}$ về hướng Nam, đồng thời cách mặt đất $50\,\mathrm{m}$; chiếc thứ hai nằm cách địa điểm xuất phát $100\,\mathrm{m}$ về hướng Tây và $100\,\mathrm{m}$ về hướng Bắc, đồng thời cách mặt đất $40\,\mathrm{m}$. Cùng thời điểm đó, một người đứng trên mặt đất quan sát thấy hai đèn trời nổi trên trời. Biết rằng, so với vị trí quan sát trên mặt đất, vị trí người quan sát có tổng khoảng cách đến hai đèn trời là nhỏ nhất. Tính khoảng cách từ vị trí người quan sát đến địa điểm thả đèn trời (đơn vị mét) (kết quả làm tròn đến hàng đơn vị).
        \end{vd}
        \noindent \underline{\textbf{Nhận xét của tác giả:}}
        Nếu ở \textbf{Ví dụ 2.2.2}, hệ tọa độ được sử dụng chủ yếu để xác định vị trí của hai Flycam và tính khoảng cách giữa chúng, thì ví dụ này khai thác hệ tọa độ theo một hướng khác: tìm vị trí trên mặt đất sao cho tổng khoảng cách từ vị trí đó đến hai chiếc đèn trời là nhỏ nhất. Tình huống thả đèn trời trong thực tế tạo ra một bài toán trực quan về khoảng cách và sự đối xứng qua mặt phẳng. Đây là một ví dụ phù hợp để minh họa cách kết hợp tọa độ, vectơ và phép đối xứng để giải quyết bài toán cực trị hình học trong không gian.
        
        \noindent \underline{\textbf{Ý tưởng giải:}}
        Lựa chọn hệ trục tọa độ $Oxyz$ phù hợp với mặt đất và các phương Đông, Tây, Nam, Bắc để xác định tọa độ của hai chiếc đèn trời. Gọi $M$ là vị trí của người đứng trên mặt đất. Khi đó, cần tìm $M$ sao cho tổng khoảng cách $MA+MB$ nhỏ nhất. Ta lấy điểm $B'$ đối xứng với $B$ qua mặt phẳng $(Oxy)$. Vì $M$ thuộc $(Oxy)$ nên $MB=MB'$, do đó
        \[
        MA+MB=MA+MB'\geq AB'.
        \]
        Dấu bằng xảy ra khi $M$ nằm trên đoạn thẳng $AB'$. Vì vậy, ta xác định giao điểm của $AB'$ với mặt phẳng $(Oxy)$, sau đó tính khoảng cách từ giao điểm đó đến vị trí của chiếc đèn trời.
        
        \noindent \underline{\textbf{Lời giải cụ thể:}}
        Xét hệ trục tọa độ $Oxyz$ như hình vẽ.
        \begin{center}
            \includegraphics[width=0.5 \textwidth]{Hinh14..jpg}
        \end{center}
        Ta có tọa độ chiếc đèn thứ nhất là $A(200;-200;50)$ và $B(-100;100;40)$.
        \begin{center}
            \includegraphics[width=0.4 \textwidth]{Hinh15.jpg}
        \end{center}
        Người đứng trên mặt đất là điểm $M$. Lấy điểm $B'$ đối xứng với $B$ qua mặt phẳng $(Oxy)$.
        Ta có $MA+MB=MA+MB'\ge AB'$, dấu bằng xảy ra khi $M$ nằm giữa $A$ và $B'$.
        
        \noindent Vậy để $MA+MB$ đạt giá trị nhỏ nhất thì $M$ là giao điểm của $AB'$ với $(Oxy)$.
        Gọi $M_0(a;b;0)$, ta có $\overrightarrow{AM_0}=(a-200;b+200;-50)$ và $\overrightarrow{AB'}=(-300;300;-90)$.
        
        \noindent Mà $\displaystyle \overrightarrow{AM_0}\parallel\overrightarrow{AB'}\Leftrightarrow
        \frac{a-200}{-300}=\frac{b+200}{300}=\frac{5}{9}
        \Leftrightarrow
        \small \begin{cases}
        a=\dfrac{100}{3},\\
        b=-\dfrac{100}{3}.
        \end{cases}$
        
        $\Rightarrow M_0\left(\dfrac{100}{3};-\dfrac{100}{3};0\right)$.
        
        \noindent Vậy khoảng cách cần tìm là: $OM_0=\sqrt{\left(\frac{100}{3}\right)^2+\left(-\frac{100}{3}\right)^2}
        =\frac{100}{3}\sqrt{2}\approx47\,\mathrm{m}.$
        
        \noindent \underline{\textbf{Lời khuyên:}}
        Đối với những bài toán trong không gian yêu cầu tìm một vị trí trên mặt phẳng sao cho tổng khoảng cách đến hai điểm đạt giá trị nhỏ nhất, không nên vội thiết lập bài toán cực trị bằng các công thức đại số. Có thể thử nhận ra yếu tố đối xứng: phản xạ một điểm qua mặt phẳng chứa vị trí cần tìm sẽ biến tổng hai đoạn đường thành một đường gấp khúc. Khi đó, có thể sử dụng bất đẳng thức tam giác để tìm giá trị nhỏ nhất. Cách làm này thường trực quan và thuận tiện hơn trong các bài toán tương tự.
