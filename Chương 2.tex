    \subsection{Biểu thức toạ độ của tích vô hướng}
    \begin{dn}
        Nếu $\vec{u}=(x_1;y_1;z_1)$ và $\vec{v}=(x_2;y_2;z_2)$ thì $\vec{u}\cdot\vec{v}=x_1x_2+y_1y_2+z_1z_2$.
    \end{dn}
    \begin{nx}
        \begin{itemize}
            \item Nếu $\vec{a}=(x;y;z)$ thì $|\vec{a}|=\sqrt{\vec{a}\cdot\vec{a}}=\sqrt{x^2+y^2+z^2}$.
            \item Nếu $A(x_1;y_1;z_1)$ và $B(x_2;y_2;z_2)$ thì
            \[
            AB=|\overrightarrow{AB}|=\sqrt{(x_2-x_1)^2+(y_2-y_1)^2+(z_2-z_1)^2}.
            \]
            \item Với hai vectơ $\vec{u}=(x_1;y_1;z_1)$ và $\vec{v}=(x_2;y_2;z_2)$ khác vectơ $\vec{0}$, ta có:

            $\vec{u}$ và $\vec{v}$ vuông góc với nhau khi và chỉ khi $x_1x_2+y_1y_2+z_1z_2=0$.
            \item $\cos(\vec{u},\vec{v})
                =\displaystyle \frac{\vec{u}\cdot\vec{v}}{|\vec{u}|\cdot|\vec{v}|}
                =\frac{x_1x_2+y_1y_2+z_1z_2}
                {\sqrt{x_1^2+y_1^2+z_1^2}\sqrt{x_2^2+y_2^2+z_2^2}}.$
        \end{itemize}
    \end{nx}
    \subsection{Diện tích và thể tích}

        \begin{itemize}
            \item Diện tích hình bình hành $ABCD$: $S_{ABCD}=\left|[\overrightarrow{AB},\overrightarrow{AD}]\right|.$
        
            \item Diện tích tam giác $ABC$: $S_{ABC}=\frac{1}{2}\left|[\overrightarrow{AB},\overrightarrow{AC}]\right|.$
        
            \item Thể tích khối hộp $ABCD.A'B'C'D'$: $V=\left|[\overrightarrow{AB},[\overrightarrow{AD},\overrightarrow{AA'}]]\right|.$
        
            \item Thể tích tứ diện $ABCD$: $V=\frac{1}{6}\left|\overrightarrow{AB}\cdot[\overrightarrow{AC},\overrightarrow{AD}]\right|.$

Khác với các ví dụ trước chủ yếu sử dụng tọa độ để xác định khoảng cách giữa các điểm, ví dụ này yêu cầu tính độ dài của nhiều đoạn dây trang trí nằm ở những vị trí khác nhau trong không gian. Đặc biệt, điều kiện $EF\parallel AC$ tạo ra sự liên hệ giữa các đoạn thẳng trong các mặt phẳng khác nhau, qua đó minh họa khả năng sử dụng tọa độ và vectơ để mô hình hóa một bài toán hình học không gian có nhiều yếu tố ràng buộc.

    \noindent \underline{\textbf{Ý tưởng giải:}} Chọn hệ trục tọa độ $Oxyz$ phù hợp với các cạnh của hình hộp chữ nhật, từ đó xác định tọa độ các đỉnh và các điểm $M$, $N$ dựa vào những khoảng cách đã cho. Tính độ dài hai đoạn dây $NC$ và $DM$ bằng công thức khoảng cách giữa hai điểm trong không gian. Đối với đoạn $EF$, sử dụng điều kiện $EF\parallel AC$ để biểu diễn vectơ $\overrightarrow{EF}$ cùng phương với $\overrightarrow{AC}$, đồng thời dựa vào vị trí của $E$, $F$ trên các cạnh để xác định độ dài $EF$. Cuối cùng, cộng độ dài ba đoạn dây trang trí và nhân với đơn giá để tính tổng chi phí.

So với \textbf{Ví dụ 3.2.1}, khi đó ta chủ yếu tính độ dài các đoạn dây dựa trên tọa độ của các điểm, còn ở ví dụ này cần tìm vị trí của một điểm trên mặt phẳng sao cho tổng độ dài đường đi là nhỏ nhất. Bài toán qua đó minh họa rõ hơn việc kết hợp tọa độ với phép đối xứng để giải quyết bài toán cực trị về khoảng cách trong không gian.
        
    \noindent \underline{\textbf{Ý tưởng giải:}}
        Chọn hệ trục tọa độ $Oxyz$ phù hợp với các cạnh của bể cá để xác định tọa độ của các điểm $A$, $G$ và $M$. Gọi $G'$ là điểm đối xứng của $G$ qua mặt phẳng $(Oxy)$. Vì điểm $I$ nằm trên $(Oxy)$ nên $IG=IG'$, do đó
        \[
        MI+IG=MI+IG'\geq MG'.

Biết các điểm $A$, $B$ và một điểm $C$ nằm trên máng trượt lần lượt có tọa độ là $(0;5;0)$, $(5;0;2)$ và $(5;0;2)$. Độ dài máng trượt nước bằng bao nhiêu mét (làm tròn kết quả đến hàng đơn vị của mét)?
    \end{vd}

    \noindent \underline{\textbf{Nhận xét của tác giả:}} Ví dụ này được lựa chọn vì xuất phát từ một mô hình thực tế quen thuộc là máng trượt nước dạng cung tròn, qua đó cho thấy vectơ có thể được kết hợp với các kiến thức hình học khác để giải quyết một bài toán thực tế. Nếu ở các ví dụ trước, vectơ chủ yếu được sử dụng để tính khoảng cách hoặc tìm đường đi ngắn nhất trong không gian, thì ở ví dụ này, tích vô hướng của vectơ được sử dụng để xác định góc tạo bởi hai đoạn thẳng trong không gian. Từ góc tìm được, ta tiếp tục sử dụng các tính chất của đường tròn để xác định độ dài cung, từ đó tính chiều dài của máng trượt.

    \noindent \underline{\textbf{Ý tưởng giải:}} Xem máng trượt là một cung của đường tròn ngoại tiếp tam giác $ABC$. Trước hết, sử dụng tọa độ của ba điểm $A$, $B$, $C$ để tính độ dài các cạnh của tam giác, từ đó xác định bán kính đường tròn ngoại tiếp. Tiếp theo, dùng tích vô hướng của hai vectơ $\overrightarrow{CA}$ và $\overrightarrow{CB}$ để tính góc $\widehat{ACB}$. Dựa vào mối liên hệ giữa góc nội tiếp và góc ở tâm, xác định số đo cung $AB$ chứa máng trượt. Cuối cùng, áp dụng công thức độ dài cung $l=R\theta$ để tính chiều dài máng trượt.

    \noindent \underline{\textbf{Lời giải cụ thể:}}
    
        \noindent \textit{Bước 1. Tính bán kính $R$ đường tròn ngoại tiếp $\triangle ABC$}

        Máng trượt là một cung tròn có hai đầu $A,B$ và đi qua điểm $C$.
        
        Độ dài máng trượt chính là độ dài cung $\widehat{ACB}$ của đường tròn ngoại tiếp $\triangle ABC$.
        
        Ta có:
        $$AB=\sqrt{6^2+7^2+(1-5)^2}=\sqrt{101}\,(\mathrm{m}),$$
        $$AC=\sqrt{5^2+0^2+(2-5)^2}=\sqrt{34}\,(\mathrm{m}),$$
        $$BC=\sqrt{(5-6)^2+7^2+(2-1)^2}=\sqrt{51}\,(\mathrm{m}).$$
        
        Khi đó: $S=\sqrt{p(p-a)(p-b)(p-c)}
        =\displaystyle \frac{\sqrt{1670}}{2}\,(\mathrm{m}^2)
        =\frac{abc}{4R}.$
        
        Suy ra bán kính $R$ của đường tròn ngoại tiếp $\triangle ABC$ là:
        $R=\displaystyle \frac{abc}{4S}
        =\frac{\sqrt{175005}}{1670}\,(\mathrm{m}).$
        
        \noindent \textit{Bước 2. Tính góc nội tiếp chắn dây $AB$}
        
        Ta có $\overrightarrow{CA}=(-5;0;3)$, $\overrightarrow{CB}=(1;7;-1)$.
        
        Khi đó:
        $\cos\widehat{ACB}
        =\cos(\overrightarrow{CA},\overrightarrow{CB})
        =\displaystyle \frac{\overrightarrow{CA}\cdot\overrightarrow{CB}}
        {|\overrightarrow{CA}|\cdot|\overrightarrow{CB}|}
        =\frac{(-5)+0+(-3)}
        {\sqrt{34}\cdot\sqrt{51}}
        =-\frac{8}{17\sqrt{6}}.$
        
        $\Rightarrow \widehat{ACB}\approx1,76\,(\mathrm{rad}).$
        
        \noindent \textit{Bước 3. Tính độ dài cung cần tìm}
        
        Cung $\widehat{AB}$ không chứa $C$ chắn bởi góc $\widehat{ACB}$.
        
        Suy ra cung $\widehat{ACB}$ bởi dây $AB$ có số đo bằng $2\pi-2\widehat{ACB}$.
        
        Vậy độ dài máng trượt bằng
        $l=R(2\pi-2\widehat{ACB})\approx14\,(\mathrm{m}).$
